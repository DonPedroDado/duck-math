timeline.ts: `export interface StepDef { id: string; dur: number; /** Seconds from step start where a predict pause happens (0 < pause < dur). */ pause?: number }`.
player.ts: add `revealed: boolean` to PlayerState; action `{ type: 'reveal' }`. Helpers:
```ts
export function pauseTime(steps: readonly StepDef[], i: number): number | null { const p = steps[i].pause; return p === undefined ? null : stepStart(steps, i) + p; }
export function awaiting(steps, s): boolean { const pt = pauseTime(steps, s.step); return pt !== null && !s.revealed && Math.abs(s.t - pt) < 1e-9; }
export type PredictPhase = 'none' | 'before' | 'awaiting' | 'after';
export function predictPhase(steps, s): PredictPhase { const pt = pauseTime(steps, s.step); if (pt === null) return 'none'; if (s.revealed || s.t > pt + 1e-9) return 'after'; return Math.abs(s.t - pt) < 1e-9 ? 'awaiting' : 'before'; }
```
Reducer changes: `initialState` adds `revealed: true`. `tick`: `const pt = s.revealed ? null : pauseTime(steps, s.step); const cap = pt !== null && s.t < pt + 1e-9 ? pt : end; t = Math.min(s.t + dt*rate, cap); playing: t < cap`. `reveal`: `awaiting ? { ...s, revealed: true, playing: true } : s`. `play` and `toggle` (when not playing) and `next`: if `awaiting(steps, s)` return the reveal result first. `next` otherwise and `play` advancing a step set `revealed: false`; `play` restart at 0 sets `revealed: false`. `prev`, `goto`, `reset` set `revealed: true` (land staged). `seek`: `const i = locate(steps, t).i; const pt = pauseTime(steps, i); revealed: pt === null ? true : t > pt + 1e-9`. clock.svelte.ts initial literal gets `revealed: true`.
