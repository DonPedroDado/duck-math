```ts
function syncUrl(i: number) {
  const id = STEP_IDS[i];
  const url = localizeHref(stepPath(id));
  if (historyMode === 'none' || location.pathname === url) { historyMode = 'push'; return; }
  (historyMode === 'replace' ? replaceState : pushState)(url, { ucStep: id });
  historyMode = 'push';
}
```
3. Back/Forward: 
```ts
$effect(() => {
  const id = page.state.ucStep ?? data.step;          // initial history entry has no state -> the loaded step
  const current = untrack(() => STEP_IDS[player.state.step]);
  if (isStepId(id) && id !== current) {
    untrack(() => { historyMode = 'none'; player.dispatch({ type: 'goto', step: STEP_IDS.indexOf(id) }); });
  }
});
```
(`untrack` from 'svelte': BOTH the read of the current step AND the `dispatch` call are untracked. `dispatch` reads `player.state` inside `reduce`, so an un-untracked dispatch would subscribe the effect to `player.state` and re-run it on every rAF tick. The effect must depend only on `page.state.ucStep` / `data.step`. `goto` shows the step staged per PD-2.)
