Today `norm()` in derive.ts has dead logic (`r === TAU ? 0 : r` can never fire, because `x % TAU` is always below TAU). `DragPoint.key()` also adds a float radian step to a float theta. 1. Keyboard steps are computed on the integer-degree grid (round the current angle to whole degrees, add the step, wrap modulo 360, convert to radians once).

aria-valuenow then comes from the same `degValue()` rule that `fmtDeg` uses, so the number and the text always agree.

```ts
const WRAP_EPS = 1e-9;

/** Wrap into [0, TAU). Values within WRAP_EPS of TAU (or of 0) become exactly 0, so float drift never yields TAU - ulp. */
export function norm(theta: number): number {
	const r = ((theta % TAU) + TAU) % TAU;
	return r < WRAP_EPS || TAU - r < WRAP_EPS ? 0 : r;
}

/** Keyboard step on the integer-degree grid: round current angle to whole degrees, add deltaDeg, wrap mod 360, convert once. */
export function keyStep(theta: number, deltaDeg: number): number {
	const cur = Math.round((norm(theta) * 180) / Math.PI);
	const next = (((cur + deltaDeg) % 360) + 360) % 360;
	return (next * Math.PI) / 180;
}
```

function norm(x: number): number {
	const r = ((x % TAU) + TAU) % TAU;

/** Degrees as shown to the user: in [0, 360), integer when within 0.05 of one, else 1 decimal. Shared by fmtDeg and aria-valuenow. */
export function degValue(theta: number): number {
	const d = (norm(theta) * 180) / Math.PI;
	if (d >= 359.95) return 0;
	const nearest = Math.round(d);

}

export function fmtDeg(theta: number, locale: string): string {

}
